# M9.5 — Integration Test Suite (GraphQL Adapter)

This milestone exercises M9's GraphQL adapter against an in-repo GraphQL API. The same integration test patterns established in M2.5–M8.5 apply: response schema assertions check field presence, types, and non-null constraints — not just status codes. The integration suite now runs eight layers in CI.

---

## Overview

M9.5 delivers three things:

1. **Example GraphQL API** — a minimal Go GraphQL server in `examples/graphql-api/` with realistic queries and mutations, an SDL schema, and a deliberate schema violation for the contract and property strategies to find
2. **Integration test config** — a `backendtest.yaml` and table test cases that exercise GraphQL discovery, execution, and response schema validation
3. **CI integration** — the GraphQL API is started alongside the orders API and `bt run --adapter graphql` is executed as part of the GitHub Actions workflow

**Exit criterion:** `bt run --strategy table --adapter graphql` passes all well-formed query and mutation cases against the in-repo GraphQL API. The broken resolver's schema violation is detected and its violation field path is reported correctly. All prior CI steps continue to pass.

## Step 1 — Example GraphQL API

### Spec

The GraphQL API is a minimal but realistic Go HTTP server using `github.com/graphql-go/graphql`. It mirrors the orders domain from the REST API so the two can be compared directly.

Endpoints:
- `POST /graphql` — the GraphQL endpoint
- `GET /health` — health check (returns `{"status":"ok"}`)

Schema (SDL):

```graphql
type Query {
  order(id: ID!): Order
  orders(status: OrderStatus): [Order!]!
  health: String!
}

type Mutation {
  createOrder(input: CreateOrderInput!): Order!
  updateOrderStatus(id: ID!, status: OrderStatus!): Order!
}

type Order {
  id: ID!
  amount: Int!
  currency: String!
  description: String
  status: OrderStatus!
  createdAt: String!
}

enum OrderStatus {
  PENDING
  CONFIRMED
  SHIPPED
  DELIVERED
  CANCELLED
}

input CreateOrderInput {
  amount: Int!
  currency: String!
  description: String
}
```

Deliberate violation: the `order` resolver occasionally returns `amount` as a string (e.g. `"one hundred"`) instead of an integer when `BT_GQL_AMOUNT_BUG=1` is set. This is the schema violation M9's assertions are designed to catch.

```go
package main_test

import (
	"bytes"

func newTestServer(t *testing.T) *httptest.Server {
	t.Helper()
	return httptest.NewServer(gqlapi.NewHandler())
}

func gqlRequest(t *testing.T, srv *httptest.Server, query string, variables map[string]any) map[string]any {
	t.Helper()
	payload := map[string]any{"query": query}
	if variables != nil {
		payload["variables"] = variables
	}
	body, _ := json.Marshal(payload)
	resp, err := http.Post(srv.URL+"/graphql", "application/json", bytes.NewReader(body))
	if err != nil {
		t.Fatalf("gqlRequest: %v", err)
	}
	defer resp.Body.Close()

	var result map[string]any
	if err := json.NewDecoder(resp.Body).Decode(&result); err != nil {
		t.Fatalf("decode response: %v", err)
	}
	return result
}

func TestServer_OrdersQuery_EachOrderHasRequiredFields(t *testing.T) {
	srv := newTestServer(t)
	defer srv.Close()

	// Create one first so we have something to check.
	createResult := gqlRequest(t, srv, `
		mutation { createOrder(input: { amount: 100, currency: "GBP" }) { id } }
	`, nil)
	createData, _ := createResult["data"].(map[string]any)
	if createData["createOrder"] == nil {
		t.Fatal("createOrder returned nil — cannot proceed with orders query test")
	}

	result := gqlRequest(t, srv, `{ orders { id amount currency status createdAt } }`, nil)

	data := result["data"].(map[string]any)
	orders := data["orders"].([]any)

	for i, o := range orders {
		order, ok := o.(map[string]any)
		if !ok {
			t.Errorf("orders[%d]: expected object, got %T", i, o)
			continue
		}
		for _, field := range []string{"id", "amount", "currency", "status", "createdAt"} {
			if order[field] == nil {
				t.Errorf("orders[%d]: required field %q is absent or null", i, field)
			}
		}
		// amount must be a number (JSON float64)
		if _, ok := order["amount"].(float64); !ok {
			t.Errorf("orders[%d]: 'amount' must be a number, got %T: %v", i, order["amount"], order["amount"])
		}
		// status must be a valid enum value
		validStatuses := map[string]bool{"PENDING": true, "CONFIRMED": true, "SHIPPED": true, "DELIVERED": true, "CANCELLED": true}
		if status, ok := order["status"].(string); !ok || !validStatuses[status] {
			t.Errorf("orders[%d]: 'status' must be a valid enum value, got %v", i, order["status"])
		}
	}
}

func TestServer_AmountBugEnabled_AmountIsString(t *testing.T) {
	// With BT_GQL_AMOUNT_BUG=1, the amount field is returned as a string.
	// This test documents the known violation.
	t.Setenv("BT_GQL_AMOUNT_BUG", "1")
	srv := newTestServer(t)
	defer srv.Close()

	created := gqlRequest(t, srv, `
		mutation { createOrder(input: { amount: 75, currency: "GBP" }) { id } }
	`, nil)
	id := created["data"].(map[string]any)["createOrder"].(map[string]any)["id"].(string)

	result := gqlRequest(t, srv, `
		query { order(id: $id) { id amount } }
	`, map[string]any{"id": id})

	data := result["data"].(map[string]any)
	if data["order"] == nil {
		t.Skip("bug mode server did not return order data")
	}
	order := data["order"].(map[string]any)

	// In bug mode, amount should be a string (the violation the contract strategy finds).
	if _, isString := order["amount"].(string); !isString {
		t.Logf("Note: BT_GQL_AMOUNT_BUG may manifest differently in this build; amount type: %T", order["amount"])
	}
}
```

## Step 4 — Integration tests

`examples/graphql-api/integration/graphql_test.go`:

```go
package integration_test

import (
	"encoding/json"
	"os"
	"os/exec"
	"path/filepath"
	"testing"
)

// GraphQLRunResult mirrors the JSON report output of bt run --adapter graphql.
type GraphQLRunResult struct {
	Summary struct {
		Total  int `json:"total"`
		Passed int `json:"passed"`
		Failed int `json:"failed"`
	} `json:"summary"`
	Results []struct {
		CaseID   string `json:"case_id"`
		Passed   bool   `json:"passed"`
		Failures []struct {
			Invariant string `json:"invariant"`
			Message   string `json:"message"`
			Field     string `json:"field"`
		} `json:"failures"`
		Response struct {
			StatusCode int    `json:"status_code"`
			Body       string `json:"body"`
		} `json:"response"`
	} `json:"results"`
}

func btBinary(t *testing.T) string {
	t.Helper()
	bin, err := exec.LookPath("bt")
	if err != nil {
		path := filepath.Join("..", "..", "..", "bt")
		if _, err := os.Stat(path); err == nil {
			return path
		}
		t.Skip("bt binary not found — run 'go build -o bt ./cmd/bt' first")
	}
	return bin
}

// TestGraphQLRun_JSONOutputIsValidSchema verifies the run produces
// a JSON report that matches the declared output schema.
func TestGraphQLRun_JSONOutputIsValidSchema(t *testing.T) {
	bt := btBinary(t)

	out, err := exec.Command(bt, "run",
		"--config", "bt/backendtest.yaml",
		"--strategy", "table",
		"--output", "json",
	).Output()
	if err != nil {
		t.Logf("bt run exited non-zero: %v", err)
	}

	var result GraphQLRunResult
	if jsonErr := json.Unmarshal(out, &result); jsonErr != nil {
		t.Fatalf("bt run produced invalid JSON: %v\noutput: %s", jsonErr, out)
	}

	s := result.Summary
	if s.Total < 0 {
		t.Errorf("summary.total must be >= 0, got %d", s.Total)
	}
	if s.Passed+s.Failed != s.Total {
		t.Errorf("passed(%d) + failed(%d) must equal total(%d)", s.Passed, s.Failed, s.Total)
	}
}

// TestGraphQLRun_AllExpectedCasesPresent verifies every declared case is reported.
func TestGraphQLRun_AllExpectedCasesPresent(t *testing.T) {
	bt := btBinary(t)

	out, _ := exec.Command(bt, "run",
		"--config", "bt/backendtest.yaml",
		"--strategy", "table",

	).Output()

	var result GraphQLRunResult
	if err := json.Unmarshal(out, &result); err != nil {
		t.Fatalf("invalid JSON: %v\noutput: %s", err, out)
	}

	expectedCases := map[string]bool{
		"gql-health":                      false,

		"gql-create-order":                false,
		"gql-create-order-missing-currency": false,
		"gql-order-not-found":             false,
	}

	for _, r := range result.Results {
		expectedCases[r.CaseID] = true
	}

			t.Errorf("expected case %q in results, not found", id)
		}
	}
}

// TestGraphQLRun_FailuresHaveFieldPaths verifies that when schema validation
// fails, the reported failure names the specific field — not just "data" or "body".
func TestGraphQLRun_FailuresHaveFieldPaths(t *testing.T) {
	bt := btBinary(t)

	out, _ := exec.Command(bt, "run",
		"--config", "bt/backendtest.yaml",
		"--strategy", "table",
		"--output", "json",
	).Output()

	var result GraphQLRunResult
	if err := json.Unmarshal(out, &result); err != nil {
		t.Fatalf("invalid JSON: %v", err)
	}

	for _, r := range result.Results {
		for j, f := range r.Failures {
			if f.Invariant == "response_schema" || f.Invariant == "gql_schema" {
				if f.Field == "" {
					t.Errorf("case %s failure[%d]: Field must be non-empty (must identify the specific field path)", r.CaseID, j)
				}
				if f.Message == "" {
					t.Errorf("case %s failure[%d]: Message must be non-empty", r.CaseID, j)
				}
			}
		}
	}
}

// TestGraphQLRun_WellFormedCasesPass verifies no false positives on valid queries.
func TestGraphQLRun_WellFormedCasesPass(t *testing.T) {
	bt := btBinary(t)

	out, _ := exec.Command(bt, "run",
		"--config", "bt/backendtest.yaml",
		"--strategy", "table",
		"--output", "json",
	).Output()

	var result GraphQLRunResult
	if err := json.Unmarshal(out, &result); err != nil {
		t.Fatalf("invalid JSON: %v", err)
	}

	// These cases should always pass against a correctly functioning server.
	shouldPass := map[string]bool{
		"gql-health":        true,
		"gql-orders-empty":  true,
		"gql-create-order":  true,
		"gql-order-not-found": true,
	}

	for _, r := range result.Results {
		if !shouldPass[r.CaseID] {
			continue
		}
		if !r.Passed {
			t.Errorf("case %q expected to pass but failed. Failures:", r.CaseID)
			for _, f := range r.Failures {
				t.Errorf("  invariant=%q field=%q message=%q", f.Invariant, f.Field, f.Message)
			}
		}
	}
}

// TestGraphQLRun_AmountBugDetected verifies that running with BT_GQL_AMOUNT_BUG=1
// causes a schema violation to be reported on the 'amount' field.
func TestGraphQLRun_AmountBugDetected(t *testing.T) {
	if os.Getenv("BT_GQL_AMOUNT_BUG") != "1" {
		t.Skip("BT_GQL_AMOUNT_BUG not set — skipping bug detection test")
	}

	bt := btBinary(t)

	out, _ := exec.Command(bt, "run",
		"--config", "bt/backendtest.yaml",
		"--strategy", "table",
		"--output", "json",
	).Output()

	var result GraphQLRunResult
	if err := json.Unmarshal(out, &result); err != nil {
		t.Fatalf("invalid JSON: %v", err)
	}

	// Expect at least one failure with field path including "amount".
	found := false
	for _, r := range result.Results {
		for _, f := range r.Failures {
			if f.Field != "" && containsString(f.Field, "amount") {
				found = true
				t.Logf("Found amount violation in case %q: field=%q message=%q", r.CaseID, f.Field, f.Message)
			}
		}
	}
	if !found {
		t.Error("expected a schema violation on 'amount' field when BT_GQL_AMOUNT_BUG=1, none found")
	}
}

func containsString(s, substr string) bool {
	return len(s) >= len(substr) && (s == substr || len(s) > 0 && stringContains(s, substr))
}

func stringContains(s, sub string) bool {
	for i := 0; i <= len(s)-len(sub); i++ {
		if s[i:i+len(sub)] == sub {
			return true
		}
	}
	return false
}
```
