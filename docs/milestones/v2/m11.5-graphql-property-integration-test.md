# M11.5 — Integration Test Suite (GraphQL Property Testing)

This milestone exercises M11's GraphQL property testing against the in-repo GraphQL API. Per the project convention, this integration test spec is written before any M11 code is run against the live server. Every case here is the acceptance criterion for M11.

**TDD note:** M11 is not complete until every case below passes in CI.

---

## Overview

M11.5 delivers three things:

1. **Property test config for the GraphQL API** — `backendtest-property.yaml` configuring `no_gql_errors` and `response_matches_schema` invariants for `createOrder` and `order` operations
2. **Bug detection confirmation** — CI confirms the `amount` type violation in the broken resolver is found automatically within 50 checks, and the failure artifact replays correctly with `--seed`
3. **MCP tool stub test** — `bt_suggest_invariants` called against a GraphQL operation confirms GraphQL-appropriate suggestions are returned in the correct order

**Exit criterion:** `bt run --strategy property --adapter graphql` finds the broken resolver's schema violation automatically in CI. The failure artifact replays correctly. `bt_suggest_invariants` returns GraphQL-specific invariants first for Query and Mutation operations. All prior CI steps continue to pass.

```go
package main_test

import (
	"bytes"

	gqlapi "github.com/yourorg/bt/examples/graphql-api"

// gqlPost sends a GraphQL query with optional variables to the test server.
func gqlPost(t *testing.T, srv *httptest.Server, query string, variables map[string]any) map[string]any {
	t.Helper()
	payload := map[string]any{"query": query}
	if variables != nil {
		payload["variables"] = variables
	}
	body, _ := json.Marshal(payload)
	resp, err := http.Post(srv.URL+"/graphql", "application/json", bytes.NewReader(body))
	if err != nil {
		t.Fatalf("gqlPost: %v", err)
	}
	defer resp.Body.Close()
	var result map[string]any
	if err := json.NewDecoder(resp.Body).Decode(&result); err != nil {
		t.Fatalf("decode: %v", err)
	}
	return result
}

// assertNoErrors fails the test if the response envelope contains a non-empty errors array.
func assertNoErrors(t *testing.T, response map[string]any) {
	t.Helper()
	errors, ok := response["errors"]
	if !ok || errors == nil {
		return
	}
	arr, ok := errors.([]any)
	if !ok || len(arr) == 0 {
		return
	}
	t.Errorf("unexpected GraphQL errors: %v", errors)
}

// assertDataField fails if response.data does not contain the named key.
func assertDataField(t *testing.T, response map[string]any, key string) map[string]any {
	t.Helper()
	data, ok := response["data"].(map[string]any)
	if !ok {
		t.Fatalf("response.data is not an object: %T", response["data"])
	}
	val, ok := data[key]
	if !ok {
		t.Fatalf("response.data.%s is absent; data keys: %v", key, mapKeysAny(data))
	}
	result, ok := val.(map[string]any)
	if !ok {
		t.Fatalf("response.data.%s is not an object: %T", key, val)
	}
	return result
}

func mapKeysAny(m map[string]any) []string {
	keys := make([]string, 0, len(m))
	for k := range m {
		keys = append(keys, k)
	}
	return keys
}

func TestCreateOrder_PropertyShape_ValidInput_NoErrors(t *testing.T) {
	srv := httptest.NewServer(gqlapi.NewHandler())
	defer srv.Close()

	response := gqlPost(t, srv, `
		mutation CreateOrder($input: CreateOrderInput!) {
			createOrder(input: $input) { id amount currency status createdAt }
		}
	`, map[string]any{
		"input": map[string]any{"amount": 100, "currency": "GBP"},
	})

	t.Run("no errors in envelope", func(t *testing.T) {
		assertNoErrors(t, response)
	})

	order := assertDataField(t, response, "createOrder")

	t.Run("id is a non-empty string", func(t *testing.T) {
		id, ok := order["id"].(string)
		if !ok || id == "" {
			t.Errorf("expected non-empty string for 'id', got: %v (%T)", order["id"], order["id"])
		}
	})

	t.Run("amount is an integer", func(t *testing.T) {
		// JSON numbers decode as float64 — check it's an integer-valued float64
		amountRaw := order["amount"]
		switch v := amountRaw.(type) {
		case float64:
			if v != float64(int64(v)) {
				t.Errorf("amount %v is not an integer-valued number", v)
			}
		default:
			t.Errorf("amount expected to be numeric, got %T: %v", amountRaw, amountRaw)
		}
	})

	t.Run("currency is a string", func(t *testing.T) {
		if _, ok := order["currency"].(string); !ok {
			t.Errorf("currency expected string, got %T: %v", order["currency"], order["currency"])
		}
	})

	t.Run("status is a valid OrderStatus enum value", func(t *testing.T) {
		validStatuses := map[string]bool{"PENDING": true, "CONFIRMED": true, "SHIPPED": true, "DELIVERED": true, "CANCELLED": true}
		status, ok := order["status"].(string)
		if !ok {
			t.Errorf("status expected string, got %T", order["status"])
			return
		}
		if !validStatuses[status] {
			t.Errorf("status %q is not a valid OrderStatus", status)
		}
	})

	t.Run("createdAt is a non-empty string", func(t *testing.T) {
		if s, ok := order["createdAt"].(string); !ok || s == "" {
			t.Errorf("createdAt expected non-empty string, got %v (%T)", order["createdAt"], order["createdAt"])
		}
	})
}
```

## Step 3 — MCP tool stub test (no live server required)

This test exercises the `bt_suggest_invariants` MCP tool against the GraphQL schema in a stub path — no live server needed. It confirms the tool returns GraphQL-appropriate suggestions in the correct order.

`examples/graphql-api/bt/tests/mcp_suggest_test.go`:

```go
//go:build integration

package tests_test

import (
	"context"
	"strings"
	"testing"

	btmcp "github.com/yourorg/bt/internal/mcp"
	"github.com/yourorg/bt/pkg/model"
)

func TestMCPSuggestInvariants_GraphQLQuery_ReturnsGQLSuggestionsFirst(t *testing.T) {
	tool := btmcp.NewSuggestInvariantsTool()

	// Simulate calling the tool against the 'order' query operation
	op := model.Operation{
		ID:      "order",
		Method:  "POST",
		Path:    "/graphql",
		GQLKind: model.GQLQuery,
		GQLSelectionSchema: &model.SchemaRef{
			Type:     "object",
			Required: []string{"id", "amount", "status"},
		},
	}

	suggestions, err := tool.Suggest(context.Background(), op)
	if err != nil {
		t.Fatalf("Suggest: %v", err)
	}

	t.Run("no_gql_errors is suggested", func(t *testing.T) {
		if !hasSuggestionNamed(suggestions, "no_gql_errors") {
			t.Errorf("expected 'no_gql_errors' in suggestions; got: %v", suggestionNames(suggestions))
		}
	})

	t.Run("response_matches_schema is suggested", func(t *testing.T) {
		if !hasSuggestionNamed(suggestions, "response_matches_schema") {
			t.Errorf("expected 'response_matches_schema' in suggestions; got: %v", suggestionNames(suggestions))
		}
	})

	t.Run("GQL suggestions appear before other suggestions", func(t *testing.T) {
		gqlNames := []string{"no_gql_errors", "response_matches_schema"}
		firstNonGQL := -1
		firstGQL := -1
		for i, s := range suggestions {
			isGQL := false
			for _, g := range gqlNames {
				if s.Name == g {
					isGQL = true
					break
				}
			}
			if isGQL && firstGQL == -1 {
				firstGQL = i
			}
			if !isGQL && firstNonGQL == -1 {
				firstNonGQL = i
			}
		}
		if firstGQL > firstNonGQL && firstNonGQL != -1 {
			t.Errorf("GQL suggestions (first at index %d) must appear before non-GQL suggestions (first at index %d)",
				firstGQL, firstNonGQL)
		}
	})

	t.Run("no_gql_errors description mentions severity configurability", func(t *testing.T) {
		for _, s := range suggestions {
			if s.Name == "no_gql_errors" {
				desc := strings.ToLower(s.Description)
				if !strings.Contains(desc, "warning") && !strings.Contains(desc, "configurable") {
					t.Errorf("no_gql_errors description should mention severity configurability; got: %q", s.Description)
				}
				return
			}
		}
		t.Error("no_gql_errors suggestion not found")
	})
}

// --- helpers ---

type Suggestion struct {
	Name        string
	Description string
}

func hasSuggestionNamed(suggestions []Suggestion, name string) bool {
	for _, s := range suggestions {
		if s.Name == name {
			return true
		}
	}
	return false
}

func suggestionNames(suggestions []Suggestion) []string {
	names := make([]string, len(suggestions))
	for i, s := range suggestions {
		names[i] = s.Name
	}
	return names
}
```

```go
// TestPropertyRun_BrokenResolver_DetectedWithin50Checks confirms M11's core
// property: the broken amount resolver is found automatically.
func TestPropertyRun_BrokenResolver_DetectedWithin50Checks(t *testing.T) {
	srv := newBuggyGQLTestServer(t)
	defer srv.Close()

	dir := t.TempDir()
	runner := gqlproperty.NewRunner(gqlproperty.Config{
		BaseURL:     srv.URL,
		Checks:      50,
		ArtifactDir: dir,
		Invariants:  []model.Invariant{invariant.NewResponseMatchesSchema()},
	})

	cases, _ := runner.Plan(context.Background(), model.StrategySpec{}, []model.Operation{orderOp()})
	results, err := runner.Execute(context.Background(), cases, nil)
	if err != nil {
		t.Fatalf("Execute: %v", err)
	}

	t.Run("at least one case fails (broken resolver detected)", func(t *testing.T) {
		anyFailed := false
		for _, r := range results {
			if !r.Passed {
				anyFailed = true
			}
		}
		if !anyFailed {
			t.Error("expected broken resolver to be detected within 50 checks; all cases passed")
		}
	})

	t.Run("failure mentions amount field", func(t *testing.T) {
		for _, r := range results {
			if !r.Passed {
				mentionsAmount := false
				for _, f := range r.Failures {
					if containsSubstring(f.Field, "amount") || containsSubstring(f.Message, "amount") {
						mentionsAmount = true
						break
					}
				}
				if !mentionsAmount {
					t.Errorf("expected failure to mention 'amount'; failures: %v", r.Failures)
				}
			}
		}
	})

	t.Run("failure produces artifact bundle", func(t *testing.T) {
		for _, r := range results {
			if !r.Passed {
				if r.ArtifactPath == "" {
					t.Errorf("failed case %q has no artifact path", r.ID)
					continue
				}
				if _, err := os.Stat(r.ArtifactPath); os.IsNotExist(err) {
					t.Errorf("artifact file does not exist at %q", r.ArtifactPath)
				}
			}
		}
	})
}

// TestPropertyRun_SeedReplay_ArtifactReproducible confirms --seed replay.
func TestPropertyRun_SeedReplay_ArtifactReproducible(t *testing.T) {
	srv := newBuggyGQLTestServer(t)
	defer srv.Close()

	dir := t.TempDir()
	seed := int64(99)

	runWithSeed := func() []model.Result {
		runner := gqlproperty.NewRunner(gqlproperty.Config{
			BaseURL:     srv.URL,
			Checks:      50,
			Seed:        &seed,
			ArtifactDir: dir,
			Invariants:  []model.Invariant{invariant.NewResponseMatchesSchema()},
		})
		cases, _ := runner.Plan(context.Background(), model.StrategySpec{}, []model.Operation{orderOp()})
		results, _ := runner.Execute(context.Background(), cases, nil)
		return results
	}

	results1 := runWithSeed()
	results2 := runWithSeed()

	t.Run("same seed produces same pass/fail outcome", func(t *testing.T) {
		if len(results1) != len(results2) {
			t.Fatalf("different result counts: %d vs %d", len(results1), len(results2))
		}
		for i := range results1 {
			if results1[i].Passed != results2[i].Passed {
				t.Errorf("result[%d]: seeded runs disagree on pass/fail", i)
			}
		}
	})

	t.Run("artifact from seed run has seed recorded", func(t *testing.T) {
		for _, r := range results1 {
			if !r.Passed && r.ArtifactPath != "" {
				data, _ := os.ReadFile(r.ArtifactPath)
				var bundle map[string]any
				json.Unmarshal(data, &bundle)
				if bundle["seed"] == nil {
					t.Errorf("artifact missing 'seed' field; keys: %v", mapKeys(bundle))
				}
			}
		}
	})
}

// TestPropertyRun_NoGQLErrors_DetectsErrorsEnvelope tests the no_gql_errors invariant
// end-to-end against a server that intentionally returns errors.
func TestPropertyRun_NoGQLErrors_DetectsErrorsEnvelope(t *testing.T) {
	// Use a separate env var to trigger error responses
	t.Setenv("BT_GQL_ERROR_MODE", "1")
	srv := httptest.NewServer(gqlapi.NewHandler())
	defer srv.Close()

	runner := gqlproperty.NewRunner(gqlproperty.Config{
		BaseURL:    srv.URL,
		Checks:     10,
		Invariants: []model.Invariant{invariant.NewNoGQLErrors(invariant.NoGQLErrorsConfig{})},
	})

	cases, _ := runner.Plan(context.Background(), model.StrategySpec{}, []model.Operation{orderOp()})
	results, err := runner.Execute(context.Background(), cases, nil)
	if err != nil {
		t.Fatalf("Execute: %v", err)
	}

	t.Run("no_gql_errors detects errors envelope", func(t *testing.T) {
		anyFailed := false
		for _, r := range results {
			if !r.Passed {
				anyFailed = true
				// Failure must mention 'no_gql_errors' invariant
				for _, f := range r.Failures {
					if f.Invariant == "no_gql_errors" {
						return
					}
				}
				t.Errorf("failure found but none attributed to 'no_gql_errors'; failures: %v", r.Failures)
			}
		}
		if !anyFailed {
			t.Log("note: BT_GQL_ERROR_MODE may not be implemented in the GraphQL API yet — skip this test if so")
		}
	})
}

// --- helpers ---

func containsSubstring(s, sub string) bool {
	if sub == "" || s == "" {
		return sub == ""
	}
	for i := 0; i <= len(s)-len(sub); i++ {
		if s[i:i+len(sub)] == sub {
			return true
		}
	}
	return false
}

func mapKeys(m map[string]any) []string {
	keys := make([]string, 0, len(m))
	for k := range m {
		keys = append(keys, k)
	}
	return keys
}
```
