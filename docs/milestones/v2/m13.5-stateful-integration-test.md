# M13.5 — Integration Test Suite (Stateful Strategy)

This milestone exercises M13's stateful strategy against both in-repo APIs. Per the project convention, all integration test cases are written before M13 implementation is run against a live server. Every case here is the acceptance criterion for M13.

**TDD note:** M13 is not complete until every case below passes in CI.

---

## Overview

M13.5 delivers five things:

1. **Hand-authored flows for the orders API** — three flows covering create-retrieve, create-update, and create-cancel journeys; written as Go server shape tests before `bt` flow definitions
2. **Hand-authored flow for the GraphQL API** — create-and-query flow using the binding model
3. **Trace-generated flow verification** — confirms at least one valid flow can be generated from the M12 HAR fixture profile and executed without error
4. **Deliberately broken flow** — confirms per-step failure reporting names the correct failing step with binding values in the artifact
5. **CI pipeline** — all flows and the trace-generated run added; complete ten-layer CI pipeline confirmed

**Exit criterion:** All hand-authored flows pass in CI against both APIs. A trace-generated flow runs without error. A deliberately failing flow reports the correct failing step with binding values and response detail. All prior CI steps continue to pass.

package main_test

import (
	"bytes"
	"encoding/json"

	"testing"
	"time"

	ordersapi "github.com/yourorg/bt/examples/orders-api"

// createOrder is a helper that creates an order and returns its ID.
func createOrder(t *testing.T, srv *httptest.Server, amount int, currency string) string {
	t.Helper()
	body := fmt.Sprintf(`{"amount":%d,"currency":"%s"}`, amount, currency)
	resp, err := http.Post(srv.URL+"/orders", "application/json", bytes.NewBufferString(body))
	if err != nil {
		t.Fatalf("createOrder: %v", err)
	}
	defer resp.Body.Close()
	if resp.StatusCode != 201 {
		t.Fatalf("createOrder: expected 201, got %d", resp.StatusCode)
	}
	var result map[string]any
	json.NewDecoder(resp.Body).Decode(&result)
	id, ok := result["id"].(string)
	if !ok || id == "" {
		t.Fatalf("createOrder: no id in response: %v", result)
	}
	return id
}

// --- Create and retrieve ---

func TestFlow_CreateAndRetrieve_BindingPropagates(t *testing.T) {
	srv := httptest.NewServer(ordersapi.NewRouter())
	defer srv.Close()

	id := createOrder(t, srv, 100, "GBP")

	resp, err := http.Get(srv.URL + "/orders/" + id)
	if err != nil {
		t.Fatalf("GET /orders/%s: %v", id, err)
	}
	defer resp.Body.Close()

	t.Run("retrieve returns 200", func(t *testing.T) {
		if resp.StatusCode != 200 {
			t.Errorf("want 200, got %d", resp.StatusCode)
		}
	})

	var body map[string]any
	json.NewDecoder(resp.Body).Decode(&body)

	t.Run("retrieved order id matches created id", func(t *testing.T) {
		if body["id"] != id {
			t.Errorf("want id %q, got %v", id, body["id"])
		}
	})
	t.Run("retrieved order status is pending", func(t *testing.T) {
		if body["status"] != "pending" {
			t.Errorf("want status 'pending', got %v", body["status"])
		}
	})
	t.Run("retrieved order has all required fields", func(t *testing.T) {
		for _, field := range []string{"id", "amount", "currency", "status", "created_at"} {
			if body[field] == nil {
				t.Errorf("required field %q missing from retrieved order", field)
			}
		}
	})
}

func TestFlow_CreateAndUpdate_StatusChanges(t *testing.T) {

	t.Run("patch returns 200", func(t *testing.T) {

			t.Errorf("want 200, got %d", resp.StatusCode)
		}
	})

	t.Run("updated status is confirmed", func(t *testing.T) {

			t.Errorf("want status 'confirmed', got %v", body["status"])
		}
	})

func TestFlow_CreateAndCancel_CancelledAtPresent(t *testing.T) {

package flows_test

import (
	"context"
	"net/http/httptest"

	ordersapi "github.com/yourorg/bt/examples/orders-api"
	"github.com/yourorg/bt/internal/strategy/stateful"

func newOrdersServer(t *testing.T) *httptest.Server {
	t.Helper()
	return httptest.NewServer(ordersapi.NewRouter())
}

func runFlow(t *testing.T, srv *httptest.Server, yamlPath string) stateful.FlowRunResult {
	t.Helper()
	flow, err := loader.LoadFlowFile(yamlPath)
	if err != nil {
		t.Fatalf("LoadFlowFile %q: %v", yamlPath, err)
	}
	runner := stateful.NewRunner(stateful.Config{BaseURL: srv.URL})
	results, err := runner.Execute(context.Background(), []model.Flow{*flow}, nil)
	if err != nil {
		t.Fatalf("Execute: %v", err)
	}
	if len(results) != 1 {
		t.Fatalf("expected 1 result, got %d", len(results))
	}
	return results[0]
}

// --- create-and-retrieve ---

func TestFlow_CreateAndRetrieve_Passes(t *testing.T) {
	srv := newOrdersServer(t)
	defer srv.Close()
	result := runFlow(t, srv, "create-and-retrieve.yaml")

	t.Run("flow passes", func(t *testing.T) {
		if !result.Passed {
			t.Errorf("expected flow to pass; steps: %v", result.Steps)
		}
	})
	t.Run("create step has binding order_id", func(t *testing.T) {
		if len(result.Steps) == 0 {
			t.Fatal("no steps in result")
		}
		if result.Steps[0].Bindings["order_id"] == "" {
			t.Error("expected non-empty order_id binding after create step")
		}

	t.Run("retrieve step path has order_id injected", func(t *testing.T) {

			t.Fatal("expected at least 2 steps")

			t.Errorf("retrieve path: want %q, got %q", wantPath, result.Steps[1].Request.Path)
		}
	})
	t.Run("retrieve step status is pending", func(t *testing.T) {
		if len(result.Steps) < 2 {
			t.Fatal("expected at least 2 steps")
		}
		if len(result.Steps[1].SchemaViolations) > 0 {

	})

## Step 4 — Deliberately broken flow and per-step failure test

This is the acceptance test for M13's per-step failure reporting.

//go:build integration

package flows_test

import (
	"context"
	"net/http/httptest"
	"os"
	"path/filepath"
	"strings"
	"testing"

	ordersapi "github.com/yourorg/bt/examples/orders-api"
	"github.com/yourorg/bt/internal/strategy/stateful"
	"github.com/yourorg/bt/internal/strategy/stateful/loader"
	"github.com/yourorg/bt/pkg/model"
)

// TestBrokenFlow_WrongExpectedStatusOnStep2 verifies per-step failure reporting.
// The flow creates an order (step 1, expects 201 — will pass) then retrieves it
// expecting 201 (wrong — server returns 200). Step 2 must fail; step 1 must pass.
func TestBrokenFlow_WrongExpectedStatusOnStep2(t *testing.T) {
	srv := httptest.NewServer(ordersapi.NewRouter())
	defer srv.Close()

	const brokenFlowYAML = `
flows:
  - id: broken-retrieve-status
    steps:
      - id: create
        operation_id: CreateOrder
        input:
          method: POST
          path: /orders
          headers:
            Content-Type: application/json
          body:
            amount: 100
            currency: GBP
        expected:
          status_code: 201
        extract:
          order_id:
            from: "$.id"
            into: path
      - id: retrieve
        operation_id: GetOrder
        input:
          method: GET
          path: "/orders/{order_id}"
        expected:
          status_code: 201  # wrong: server returns 200
`

	flow, err := loader.LoadFlow(strings.NewReader(brokenFlowYAML))
	if err != nil {
		t.Fatalf("LoadFlow: %v", err)
	}

	dir := t.TempDir()
	runner := stateful.NewRunner(stateful.Config{
		BaseURL:     srv.URL,
		ArtifactDir: dir,
	})
	results, err := runner.Execute(context.Background(), []model.Flow{*flow}, nil)
	if err != nil {
		t.Fatalf("Execute: %v", err)
	}
	result := results[0]

	t.Run("flow fails overall", func(t *testing.T) {
		if result.Passed {
			t.Error("expected flow to fail (step 2 has wrong expected status)")
		}
	})

	t.Run("both steps are present in result", func(t *testing.T) {
		if len(result.Steps) != 2 {
			t.Fatalf("expected 2 step results, got %d", len(result.Steps))
		}
	})

	t.Run("step 1 (create) passes", func(t *testing.T) {
		if !result.Steps[0].Passed {
			t.Errorf("expected step 1 to pass; violations: %v, bindingFailure: %v",
				result.Steps[0].SchemaViolations, result.Steps[0].BindingFailure)
		}
	})

	t.Run("step 2 (retrieve) fails", func(t *testing.T) {
		if result.Steps[1].Passed {
			t.Error("expected step 2 to fail (wrong expected status)")
		}
	})

	t.Run("step 2 has correct status code in result", func(t *testing.T) {
		if result.Steps[1].StatusCode != 200 {
			t.Errorf("step 2 actual status: want 200, got %d", result.Steps[1].StatusCode)
		}
	})

	t.Run("step 1 binding is visible in step 2 result", func(t *testing.T) {
		// The order_id binding must be present in step 2's resolved request path
		if !strings.Contains(result.Steps[1].Request.Path, "/orders/") {
			t.Errorf("step 2 path should contain '/orders/{id}': got %q", result.Steps[1].Request.Path)
		}
	})

	t.Run("artifact is written", func(t *testing.T) {
		if result.ArtifactPath == "" {
			t.Fatal("expected artifact path to be set")
		}
		if _, err := os.Stat(result.ArtifactPath); os.IsNotExist(err) {
			t.Errorf("artifact file not found at %q", result.ArtifactPath)
		}
	})
}

// TestBrokenFlow_ArtifactContainsPerStepDetail verifies the artifact format.
func TestBrokenFlow_ArtifactContainsPerStepDetail(t *testing.T) {
	srv := httptest.NewServer(ordersapi.NewRouter())
	defer srv.Close()

	const brokenFlowYAML = `
flows:
  - id: artifact-test
    steps:
      - id: create
        operation_id: CreateOrder
        input:
          method: POST
          path: /orders
          headers:
            Content-Type: application/json
          body:
            amount: 100
            currency: GBP
        expected:
          status_code: 201
        extract:
          order_id: {from: "$.id", into: path}
      - id: retrieve
        operation_id: GetOrder
        input:
          method: GET
          path: "/orders/{order_id}"
        expected:
          status_code: 999  # deliberately wrong
`

	flow, _ := loader.LoadFlow(strings.NewReader(brokenFlowYAML))
	dir := t.TempDir()
	runner := stateful.NewRunner(stateful.Config{BaseURL: srv.URL, ArtifactDir: dir})
	results, _ := runner.Execute(context.Background(), []model.Flow{*flow}, nil)
	result := results[0]

	if result.ArtifactPath == "" {
		t.Skip("no artifact written — cannot verify artifact format")
	}

	data, err := os.ReadFile(result.ArtifactPath)
	if err != nil {
		t.Fatalf("read artifact: %v", err)
	}

	var artifact map[string]any
	if err := json.Unmarshal(data, &artifact); err != nil {
		t.Fatalf("artifact is not valid JSON: %v", err)
	}

	t.Run("artifact has flow_id", func(t *testing.T) {
		if artifact["flow_id"] == nil || artifact["flow_id"] == "" {
			t.Error("artifact missing flow_id")
		}
	})

	t.Run("artifact has steps array", func(t *testing.T) {
		steps, ok := artifact["steps"].([]any)
		if !ok || len(steps) == 0 {
			t.Error("artifact steps must be a non-empty array")
		}
	})

	steps, _ := artifact["steps"].([]any)
	if len(steps) >= 1 {
		step0, _ := steps[0].(map[string]any)
		t.Run("step 1 has step_id", func(t *testing.T) {
			if step0["step_id"] == nil {
				t.Error("step 0 missing step_id")
			}
		})
		t.Run("step 1 has bindings", func(t *testing.T) {
			// Bindings may be empty map for step 1, but key must be present
			if _, ok := step0["bindings"]; !ok {
				t.Error("step 0 missing bindings key")
			}
		})
		t.Run("step 1 has request", func(t *testing.T) {
			if step0["request"] == nil {
				t.Error("step 0 missing request")
			}
		})
		t.Run("step 1 has response", func(t *testing.T) {
			if step0["response"] == nil {
				t.Error("step 0 missing response")
			}
		})
	}

	if len(steps) >= 2 {
		step1, _ := steps[1].(map[string]any)
		t.Run("step 2 request path contains injected order_id", func(t *testing.T) {
			req, _ := step1["request"].(map[string]any)
			path, _ := req["path"].(string)
			if !strings.Contains(path, "/orders/") {
				t.Errorf("step 2 request path should contain injected id: got %q", path)
			}
		})
	}
}

//go:build integration

package flows_test

import (
	"context"
	"net/http/httptest"
	"testing"

	ordersapi "github.com/yourorg/bt/examples/orders-api"
	statefulgen "github.com/yourorg/bt/internal/strategy/stateful/gen"
	"github.com/yourorg/bt/internal/strategy/stateful"
	"github.com/yourorg/bt/pkg/model"
)

// TestTraceGeneratedFlow_RunsWithoutError verifies the M13 exit criterion:
// a flow generated from the M12 trace profile executes without error.
func TestTraceGeneratedFlow_RunsWithoutError(t *testing.T) {
	// Load the M12 HAR fixture profile
	profile, err := model.ParseProfile("../../trace/profile.json")
	if err != nil {
		t.Skipf("trace profile not found — run bt trace import first: %v", err)
	}
	if profile.Sequences == nil || len(profile.Sequences.StartProbability) < 2 {
		t.Skip("trace profile has insufficient sequence data for flow generation")
	}

	flows := statefulgen.GenerateFlows(profile, statefulgen.GenerateFlowsConfig{
		Count:    5,
		MaxSteps: 3,

//go:build integration

package flows_test

import (
	"context"
	"net/http/httptest"
	"strings"
	"sync/atomic"
	"testing"

	ordersapi "github.com/yourorg/bt/examples/orders-api"
	"github.com/yourorg/bt/internal/strategy/stateful"
	"github.com/yourorg/bt/internal/strategy/stateful/loader"
	"github.com/yourorg/bt/pkg/model"

func TestReplay_UsesSavedBindings_NotServerValues(t *testing.T) {
	// First execution: server returns ord_FIRST for the create step.
	// Replay: server returns ord_SECOND for the create step.
	// Replay must use ord_FIRST from the artifact — not ord_SECOND.

	var callCount int32
	var firstID string

	srv := httptest.NewServer(ordersapi.NewRouter())
	defer srv.Close()

	// We need the real server to actually create the order so the GET works.
	// We don't control the ID, but we capture it from step 1's binding.
	const failOnRetrieve = `
flows:
  - id: replay-test
    steps:
      - id: create
        operation_id: CreateOrder
        input:
          method: POST
          path: /orders
          headers:
            Content-Type: application/json
          body:
            amount: 100
            currency: GBP
        expected:
          status_code: 201
        extract:
          order_id: {from: "$.id", into: path}
      - id: retrieve
        operation_id: GetOrder
        input:
          method: GET
          path: "/orders/{order_id}"
        expected:
          status_code: 999  # wrong — forces artifact creation
`

	flow, _ := loader.LoadFlow(strings.NewReader(failOnRetrieve))
	dir := t.TempDir()
	runner := stateful.NewRunner(stateful.Config{BaseURL: srv.URL, ArtifactDir: dir})

	// First run
	results, _ := runner.Execute(context.Background(), []model.Flow{*flow}, nil)
	result := results[0]

	if result.ArtifactPath == "" {
		t.Skip("no artifact written — cannot test replay")
	}
	if len(result.Steps) == 0 {
		t.Skip("no steps in result — cannot test replay")
	}

	// Capture the order_id binding from the first run
	firstID, _ = result.Steps[0].Bindings["order_id"].(string)
	if firstID == "" {
		t.Skip("no order_id binding captured — cannot test replay")
	}
	_ = callCount

	// Replay
	replayResult, err := runner.Replay(context.Background(), result.ArtifactPath)
	if err != nil {
		t.Fatalf("Replay: %v", err)
	}

	t.Run("replay has same number of steps as original", func(t *testing.T) {
		if len(replayResult.Steps) != len(result.Steps) {
			t.Errorf("original had %d steps, replay has %d", len(result.Steps), len(replayResult.Steps))
		}
	})

	t.Run("replay uses saved order_id binding", func(t *testing.T) {
		if len(replayResult.Steps) < 2 {
			t.Skip("fewer than 2 steps in replay")
		}
		wantPath := "/orders/" + firstID
		if replayResult.Steps[1].Request.Path != wantPath {
			t.Errorf("replay step 2 path: want %q (saved binding), got %q",
				wantPath, replayResult.Steps[1].Request.Path)
		}
	})
}
