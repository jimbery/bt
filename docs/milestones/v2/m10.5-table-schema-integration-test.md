# M10.5 — Integration Test Suite (Table Schema Assertions)

This milestone exercises M10's schema assertion capability against both in-repo APIs — the orders API and the GraphQL API. Per the project convention, the integration test config and CI steps defined here are written **before** M10 implementation is run against a live server. The test cases here define exactly what "field-level schema violation reporting" means in a real end-to-end run.

**TDD note:** the test cases in this milestone are the acceptance criterion for M10. M10 is not complete until every case below passes in CI.

---

## Overview

M10.5 delivers three things:

1. **Updated orders API table cases** — every operation now declares `expected.schema` with the full response shape inline. At least one case per operation exercises a schema assertion.
2. **Updated GraphQL API table cases** — every mutation and query operation declares `gql_data_schema` where a response body is returned.
3. **A deliberately wrong schema case** — a case whose `expected.schema` is intentionally misconfigured, confirmed to fail with the exact field path, expected type, actual value, and severity in the run report. This is the acceptance test for the quality of violation reporting.

**Exit criterion:** All schema-asserting table cases pass in CI against both APIs. The deliberately wrong schema case fails with a violation report that names the specific field, expected type, and actual value — not just "schema mismatch". All prior CI steps continue to pass.

```go
// assertField fails the test if the field is absent from the map.
func assertField(t *testing.T, body map[string]any, field string) {
	t.Helper()
	if _, ok := body[field]; !ok {
		t.Errorf("expected field %q in response body, got keys: %v", field, mapKeys(body))
	}
}

// assertFieldType fails the test if the field has the wrong JSON type.
func assertFieldType(t *testing.T, body map[string]any, field string, wantType string) {
	t.Helper()
	val, ok := body[field]
	if !ok {
		t.Errorf("field %q absent from response body", field)
		return
	}
	gotType := jsonTypeName(val)
	if gotType != wantType {
		t.Errorf("field %q: expected type %q, got %q (value: %v)", field, wantType, gotType, val)
	}
}

// assertEnumValue fails if the field value is not in the allowed set.
func assertEnumValue(t *testing.T, body map[string]any, field string, allowed []string) {
	t.Helper()
	val, ok := body[field]
	if !ok {
		t.Errorf("field %q absent", field)
		return
	}
	s, ok := val.(string)
	if !ok {
		t.Errorf("field %q: expected string for enum check, got %T", field, val)
		return
	}
	for _, a := range allowed {
		if s == a {
			return
		}
	}
	t.Errorf("field %q: value %q not in allowed set %v", field, s, allowed)
}

func mapKeys(m map[string]any) []string {
	keys := make([]string, 0, len(m))
	for k := range m {
		keys = append(keys, k)
	}
	return keys
}

func jsonTypeName(v any) string {
	switch v.(type) {
	case string:
		return "string"
	case float64:
		return "number"
	case bool:
		return "boolean"
	case nil:
		return "null"
	case map[string]any:
		return "object"
	case []any:
		return "array"
	default:
		return "unknown"
	}
}

// GetOrder response shape

func TestGetOrder_SchemaShape(t *testing.T) {
	srv := httptest.NewServer(ordersapi.NewRouter())
	defer srv.Close()

	// Create an order first
	payload := `{"amount": 50, "currency": "USD"}`
	createResp, _ := http.Post(srv.URL+"/orders", "application/json", bytes.NewBufferString(payload))
	defer createResp.Body.Close()
	var created map[string]any
	json.NewDecoder(createResp.Body).Decode(&created)
	id := created["id"].(string)

	resp, err := http.Get(srv.URL + "/orders/" + id)
	if err != nil {
		t.Fatalf("request failed: %v", err)
	}
	defer resp.Body.Close()

	var body map[string]any
	if err := json.NewDecoder(resp.Body).Decode(&body); err != nil {
		t.Fatalf("response is not valid JSON: %v", err)
	}

	requiredFields := []struct {
		name     string
		jsonType string
	}{
		{"id", "string"},
		{"amount", "number"},
		{"currency", "string"},
		{"status", "string"},
		{"created_at", "string"},
	}

	for _, f := range requiredFields {
		t.Run(f.name+" field present and correct type", func(t *testing.T) {
			assertField(t, body, f.name)
			assertFieldType(t, body, f.name, f.jsonType)
		})
	}

	t.Run("id matches the requested id", func(t *testing.T) {
		if body["id"] != id {
			t.Errorf("expected id %q, got %v", id, body["id"])
		}
	})
}
```

## Step 3 — Violation acceptance test runner

The deliberately wrong schema cases must be verified to produce the correct violation detail. This is done with a small Go integration test that starts the orders API, runs the bad-schema case, and asserts the violation report shape.

`examples/orders-api/bt/tests/violation_acceptance_test.go`:

```go
//go:build integration

package tests_test

import (
	"context"
	"net/http/httptest"
	"strings"
	"testing"

	ordersapi "github.com/yourorg/bt/examples/orders-api"
	"github.com/yourorg/bt/internal/strategy/table"
)

// TestSchemaViolationAcceptance verifies that the deliberately wrong schema
// case produces a violation report with the exact required detail.
// Run with: go test -tags integration ./examples/orders-api/bt/tests/...
func TestSchemaViolationAcceptance(t *testing.T) {
	srv := httptest.NewServer(ordersapi.NewRouter())
	defer srv.Close()

	// Load just the acceptance test case
	caseYAML := `
cases:
  - id: schema-violation-acceptance-test
    operation_id: CreateOrder
    input:
      method: POST
      path: /orders
      headers:
        Content-Type: application/json
      body:
        amount: 100
        currency: GBP
    expected:
      status_code: 201
      schema:
        type: object
        required: [id, amount, currency, status, created_at]
        properties:
          id:         { type: string }
          amount:     { type: string }   # intentionally wrong
          currency:   { type: string }
          status:     { type: string }
          created_at: { type: string }
`
	cases, err := table.LoadCasesFromReader(strings.NewReader(caseYAML))
	if err != nil {
		t.Fatalf("LoadCasesFromReader: %v", err)
	}

	strat := table.New()
	runner := table.NewHTTPRunner(srv.URL)
	results, err := strat.Execute(context.Background(), cases, runner)
	if err != nil {
		t.Fatalf("Execute: %v", err)
	}
	if len(results) != 1 {
		t.Fatalf("expected 1 result, got %d", len(results))
	}

	result := results[0]

	t.Run("case fails (schema violation present)", func(t *testing.T) {
		if result.Passed {
			t.Error("expected the acceptance test case to fail; the schema has an intentional violation")
		}
	})

	t.Run("violation is on field $.amount", func(t *testing.T) {
		found := false
		for _, v := range result.SchemaViolations {
			if v.Field == "$.amount" {
				found = true
				break
			}
		}
		if !found {
			t.Errorf("expected violation on '$.amount'; violations: %v", result.SchemaViolations)
		}
	})

	t.Run("violation reports expected type as 'string'", func(t *testing.T) {
		for _, v := range result.SchemaViolations {
			if v.Field == "$.amount" {
				if v.ExpectedType != "string" {
					t.Errorf("ExpectedType: want 'string', got %q", v.ExpectedType)
				}
			}
		}
	})

	t.Run("violation reports actual type — not just 'schema mismatch'", func(t *testing.T) {
		for _, v := range result.SchemaViolations {
			if v.Field == "$.amount" {
				if v.ActualType == "" || v.ActualType == "schema mismatch" {
					t.Errorf("ActualType must be a descriptive type name, got %q", v.ActualType)
				}
			}
		}
	})

	t.Run("violation severity is Critical", func(t *testing.T) {
		for _, v := range result.SchemaViolations {
			if v.Field == "$.amount" {
				if v.Severity != table.SeverityCritical {
					t.Errorf("Severity: want Critical, got %v", v.Severity)
				}
			}
		}
	})

	t.Run("violation message is not empty or generic", func(t *testing.T) {
		for _, v := range result.SchemaViolations {
			if v.Field == "$.amount" {
				if v.Message == "" || v.Message == "schema mismatch" {
					t.Errorf("Message must be descriptive, got %q", v.Message)
				}
			}
		}
	})
}
```
