# M12.5 — Integration Test Suite (Trace Adapter)

This milestone exercises M12's trace adapter against the in-repo orders API. Per the project convention, all integration test cases are written before M12 implementation is run against a live server. Every case here is the acceptance criterion for M12.

**TDD note:** M12 is not complete until every case below passes in CI.

---

## Overview

M12.5 delivers four things:

1. **HAR fixture** — a committed HAR file representing realistic orders API traffic, with known distributions that the integration tests assert against
2. **Profile correctness tests** — Go tests that parse the generated profile and assert every structural field is populated correctly
3. **Distribution verification** — a property-based integration test that runs the property strategy with the trace profile and confirms generated `currency` values match the HAR distribution over 500 draws
4. **CI pipeline** — `bt trace import` and `bt trace inspect` steps added; distribution test confirmed; property run with profile verified

**Exit criterion:** `bt trace import examples/orders-api/bt/trace/sample.har` produces a valid profile in CI. A property run with the profile configured generates `currency` values with GBP frequency >60% over 500 draws. All prior CI steps continue to pass.

---

## Step 1 — HAR fixture

The HAR fixture is committed at `examples/orders-api/bt/trace/sample.har`. It is generated once by running a scripted session against the orders API test server and capturing requests/responses programmatically — not by hand.

The fixture is generated by the test below. Once generated it is committed and treated as stable — it is not regenerated on every CI run.

### Fixture generation test

`examples/orders-api/bt/trace/generate_fixture_test.go`:

```go
//go:build generate_fixture

package trace_test

import (
	"bytes"
	"encoding/json"
	"fmt"
	"net/http"
	"net/http/httptest"
	"os"
	"testing"
	"time"

	ordersapi "github.com/yourorg/bt/examples/orders-api"
)

// TestGenerateHARFixture generates sample.har by scripting realistic traffic
// against the orders API test server. Run with:
//   go test -tags generate_fixture ./examples/orders-api/bt/trace/... -run TestGenerateHARFixture -v
//
// The output is committed as examples/orders-api/bt/trace/sample.har.
// Do not run this in CI — the fixture is committed, not generated on the fly.
func TestGenerateHARFixture(t *testing.T) {
	srv := httptest.NewServer(ordersapi.NewRouter())
	defer srv.Close()

	recorder := newHARRecorder()

	client := &http.Client{}

	makeRequest := func(method, path, body string) *http.Response {
		t.Helper()
		var req *http.Request
		var err error
		if body != "" {
			req, err = http.NewRequest(method, srv.URL+path, bytes.NewBufferString(body))
			req.Header.Set("Content-Type", "application/json")
		} else {
			req, err = http.NewRequest(method, srv.URL+path, nil)
		}
		if err != nil {
			t.Fatalf("newRequest: %v", err)
		}
		start := time.Now()
		resp, err := client.Do(req)
		if err != nil {
			t.Fatalf("Do: %v", err)
		}
		elapsed := time.Since(start)
		recorder.Record(req, resp, start, elapsed)
		return resp
	}

	// Session distribution:
	// - 70 POST /orders with GBP (amount 10-200)
	// - 20 POST /orders with USD (amount 50-300)
	// - 10 POST /orders with EUR (amount 100-500)
	// - Each CreateOrder is followed by GET /orders/{id} (~90% of the time)
	// - 20 GET /orders with ?status=pending

	var orderIDs []string

	// GBP orders
	for i := 0; i < 70; i++ {
		amount := 10 + (i * 3 % 190)
		body := fmt.Sprintf(`{"amount":%d,"currency":"GBP"}`, amount)
		resp := makeRequest("POST", "/orders", body)
		if resp.StatusCode == 201 {
			var created map[string]any
			json.NewDecoder(resp.Body).Decode(&created)
			resp.Body.Close()
			if id, ok := created["id"].(string); ok {
				orderIDs = append(orderIDs, id)
			}
		} else {
			resp.Body.Close()
		}
	}

	// USD orders
	for i := 0; i < 20; i++ {
		amount := 50 + (i * 13 % 250)
		body := fmt.Sprintf(`{"amount":%d,"currency":"USD"}`, amount)
		resp := makeRequest("POST", "/orders", body)
		if resp.StatusCode == 201 {
			var created map[string]any
			json.NewDecoder(resp.Body).Decode(&created)
			resp.Body.Close()
			if id, ok := created["id"].(string); ok {
				orderIDs = append(orderIDs, id)
			}
		} else {
			resp.Body.Close()
		}
	}

	// EUR orders
	for i := 0; i < 10; i++ {
		amount := 100 + (i * 40 % 400)
		body := fmt.Sprintf(`{"amount":%d,"currency":"EUR"}`, amount)
		resp := makeRequest("POST", "/orders", body)
		if resp.StatusCode == 201 {
			var created map[string]any
			json.NewDecoder(resp.Body).Decode(&created)
			resp.Body.Close()
			if id, ok := created["id"].(string); ok {
				orderIDs = append(orderIDs, id)
			}
		} else {
			resp.Body.Close()
		}
	}

	// GET requests for 90% of created orders
	for i, id := range orderIDs {
		if i%10 == 9 {
			continue // skip 10%
		}
		resp := makeRequest("GET", "/orders/"+id, "")
		resp.Body.Close()
	}

	// List requests with status filter
	for i := 0; i < 20; i++ {
		resp := makeRequest("GET", "/orders?status=pending", "")
		resp.Body.Close()
	}

	// Write HAR file
	harData := recorder.ToHAR()
	data, err := json.MarshalIndent(harData, "", "  ")
	if err != nil {
		t.Fatalf("marshal HAR: %v", err)
	}

	outputPath := "sample.har"
	if err := os.WriteFile(outputPath, data, 0644); err != nil {
		t.Fatalf("write HAR: %v", err)
	}

	t.Logf("Generated %s with %d entries", outputPath, len(recorder.entries))
	t.Logf("Operations: %d CreateOrder, %d GetOrder, %d ListOrders",
		recorder.countMethod("POST", "/orders"),
		recorder.countMethod("GET", "/orders/{id}"),
		recorder.countMethod("GET", "/orders"),
	)
}
```

### Fixture contents — known distributions

The committed fixture must contain:

| Field | Expected distribution |
|-------|-----------------------|
| `currency` in CreateOrder | GBP ≈ 70%, USD ≈ 20%, EUR ≈ 10% |
| `amount` in CreateOrder | min ≈ 10, max ≈ 500, range non-nil |
| CreateOrder call count | ≥ 90 entries |
| GetOrder call count | ≥ 80 entries |
| ListOrders call count | ≥ 20 entries |
| CreateOrder → GetOrder transition probability | > 0.80 |

These distributions are the ground truth that the profile correctness tests below assert.

## Step 3 — Distribution verification test

This is the core M12 acceptance test: run the property strategy with the trace profile over 500 draws and assert the `currency` distribution matches the HAR-observed distribution within a statistical tolerance.

`examples/orders-api/bt/trace/distribution_test.go`:

```go
//go:build integration

package trace_test

import (
	"bytes"
	"context"
	"encoding/json"
	"fmt"
	"net/http"
	"net/http/httptest"
	"path/filepath"
	"sync"
	"testing"

	ordersapi "github.com/yourorg/bt/examples/orders-api"
	"github.com/yourorg/bt/internal/strategy/property"
	"github.com/yourorg/bt/pkg/model"
)

// TestDistributionVerification is the M12 exit criterion test:
// the property strategy with the trace profile generates currency values
// matching the HAR-observed distribution over 500 draws.
func TestDistributionVerification_TraceProfile_CurrencyMatchesHARDistribution(t *testing.T) {
	profile := importFixture(t)

	// Record all request bodies
	var mu sync.Mutex
	var bodies [][]byte
	srv := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		if r.Method == "POST" && r.URL.Path == "/orders" {
			var buf bytes.Buffer
			buf.ReadFrom(r.Body)
			mu.Lock()
			bodies = append(bodies, buf.Bytes())
			mu.Unlock()
			w.Header().Set("Content-Type", "application/json")
			w.WriteHeader(201)
			w.Write([]byte(`{"id":"ord_1","amount":100,"currency":"GBP","status":"pending","created_at":"2024-01-15T10:00:00Z"}`))
		} else {
			w.Header().Set("Content-Type", "application/json")
			w.WriteHeader(200)
			w.Write([]byte(`{"orders":[],"total":0}`))
		}
	}))
	defer srv.Close()

	runner := property.NewRunner(property.Config{
		BaseURL:      srv.URL,
		Checks:       500,
		TraceProfile: profile,
	})

	op := createOrderOpFromSpec(t) // load from the committed OpenAPI spec
	cases, err := runner.Plan(context.Background(), model.StrategySpec{}, []model.Operation{op})
	if err != nil {
		t.Fatalf("Plan: %v", err)
	}
	runner.Execute(context.Background(), cases, nil)

	// Extract currency values
	currencies := extractCurrenciesFromBodies(t, bodies)

	t.Run("at least 400 currency values captured", func(t *testing.T) {
		if len(currencies) < 400 {
			t.Errorf("expected ≥400 currency values, got %d (some requests may not be POST /orders)", len(currencies))
		}
	})

	if len(currencies) == 0 {
		t.Skip("no currency values captured — cannot verify distribution")
	}

	gbpCount := 0
	usdCount := 0
	eurCount := 0
	otherCount := 0
	for _, c := range currencies {
		switch c {
		case "GBP":
			gbpCount++
		case "USD":
			usdCount++
		case "EUR":
			eurCount++
		default:
			otherCount++
		}
	}

	total := float64(len(currencies))
	gbpFreq := float64(gbpCount) / total
	usdFreq := float64(usdCount) / total
	eurFreq := float64(eurCount) / total

	t.Logf("Distribution over %d draws: GBP=%.2f USD=%.2f EUR=%.2f other=%.2f",
		len(currencies), gbpFreq, usdFreq, eurFreq, float64(otherCount)/total)

	t.Run("GBP frequency is >60% (profile says ~70%)", func(t *testing.T) {
		if gbpFreq < 0.60 {
			t.Errorf("GBP frequency: want >0.60, got %.2f", gbpFreq)
		}
	})

	t.Run("USD frequency is >10% (profile says ~20%)", func(t *testing.T) {
		if usdFreq < 0.10 {
			t.Errorf("USD frequency: want >0.10, got %.2f", usdFreq)
		}
	})

	t.Run("no currency values outside GBP/USD/EUR", func(t *testing.T) {
		if otherCount > 0 {
			t.Errorf("expected only GBP/USD/EUR; got %d unexpected values", otherCount)
		}
	})

	t.Run("GBP is the most frequent currency", func(t *testing.T) {
		if gbpFreq <= usdFreq || gbpFreq <= eurFreq {
			t.Errorf("GBP (%.2f) must be the most frequent; USD=%.2f EUR=%.2f", gbpFreq, usdFreq, eurFreq)
		}
	})
}

func TestDistributionVerification_WithoutProfile_GeneratesUniformCurrency(t *testing.T) {
	// Without a profile, the generator produces arbitrary strings for currency —
	// not weighted towards GBP/USD/EUR. This confirms the profile is the source
	// of the distribution, not some other mechanism.

	var mu sync.Mutex
	var bodies [][]byte
	srv := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		if r.Method == "POST" && r.URL.Path == "/orders" {
			var buf bytes.Buffer
			buf.ReadFrom(r.Body)
			mu.Lock()
			bodies = append(bodies, buf.Bytes())
			mu.Unlock()
		}
		w.Header().Set("Content-Type", "application/json")
		w.WriteHeader(201)
		w.Write([]byte(`{"id":"ord_1","amount":100,"currency":"GBP","status":"pending","created_at":"2024-01-15T10:00:00Z"}`))
	}))
	defer srv.Close()

	runner := property.NewRunner(property.Config{
		BaseURL:      srv.URL,
		Checks:       200,
		TraceProfile: nil, // no profile
	})

	op := createOrderOpFromSpec(t)
	cases, _ := runner.Plan(context.Background(), model.StrategySpec{}, []model.Operation{op})
	runner.Execute(context.Background(), cases, nil)

	currencies := extractCurrenciesFromBodies(t, bodies)

	if len(currencies) == 0 {
		t.Skip("no currencies captured")
	}

	gbpCount := 0
	for _, c := range currencies {
		if c == "GBP" {
			gbpCount++
		}
	}
	gbpFreq := float64(gbpCount) / float64(len(currencies))

	t.Run("without profile GBP frequency is not artificially high", func(t *testing.T) {
		// With no profile, generator produces arbitrary strings.
		// GBP won't appear at all (schema-derived string generator doesn't know about GBP).
		// This test confirms no profile → no distribution bias.
		if gbpFreq > 0.50 {
			t.Errorf("without profile, GBP frequency should not be >50%%; got %.2f — is the profile being used unexpectedly?", gbpFreq)
		}
	})
}

func extractCurrenciesFromBodies(t *testing.T, bodies [][]byte) []string {
	t.Helper()
	var currencies []string
	for _, body := range bodies {
		var m map[string]any
		if err := json.Unmarshal(body, &m); err != nil {
			continue
		}
		if c, ok := m["currency"].(string); ok && c != "" {
			currencies = append(currencies, c)
		}
	}
	return currencies
}

func extractAmountsFromBodies(t *testing.T, bodies [][]byte) []int {
	t.Helper()
	var amounts []int
	for _, body := range bodies {
		var m map[string]any
		if err := json.Unmarshal(body, &m); err != nil {
			continue
		}
		if a, ok := m["amount"].(float64); ok {
			amounts = append(amounts, int(a))
		}
	}

}

func minSlice(s []int) int {
	if len(s) == 0 {
		return 0
	}
	min := s[0]
	for _, v := range s[1:] {
		if v < min {
			min = v
		}
	}
	return min
}

		return 0
	}
	max := s[0]
	for _, v := range s[1:] {
		if v > max {
			max = v
		}
	}
	return max

			}
```
