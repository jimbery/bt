# M8.5 — Integration Test Suite (Contract Strategy + CI Hardening)

This milestone exercises M8's contract strategy and CI tooling against the in-repo orders API. Every assertion checks the full response schema — field presence, types, nullability, and enum values — not just status codes. The integration suite now runs seven layers in CI: table (M2.5), replay (M3.5), property (M4.5), fuzz (M5.5), MCP (M6.5), AI scaffold (M7.5), and contract (M8.5).

---

## Overview

M8.5 delivers four things:

1. **Contract test config** — a `backendtest.yaml` for the contract strategy against the orders API, with response schema assertions for every declared operation
2. **Baseline fixture** — a `.bt/baseline.yaml` that quarantines the known-broken endpoint (`GetOrderBroken`), demonstrating the quarantine model in a real run
3. **`bt doctor` integration test** — a Go test that runs `bt doctor` against the orders API environment and verifies the structured output shape
4. **CI integration** — the contract step runs in GitHub Actions, verifies exit codes, and uploads violation reports as JUnit annotations

**Exit criterion:** `bt run --strategy contract` passes against all well-formed orders API endpoints. `GetOrderBroken` is quarantined and does not affect the exit code, but its violations are visible in the report. `bt doctor` passes all five checks in CI. Exit code `0` is produced. All prior CI steps continue to pass.

## Step 1 — Orders API: contract-visible rough edges

The orders API already has `GET /orders/{id}/broken`, which returns an inconsistent response shape. M8.5 adds one more contract-testable behaviour: a response field type inconsistency on `PATCH /orders/{id}` when the order transitions to `cancelled` — the `cancelled_at` field is sometimes returned as an integer timestamp instead of a `date-time` string. This is a real-world class of contract violation.

### Spec

Extend the orders API with the following behaviour:

- `PATCH /orders/{id}` with `{"status": "cancelled"}` sometimes returns `cancelled_at` as an integer Unix timestamp instead of an ISO 8601 string (controlled by `ORDERS_API_TIMESTAMP_BUG=1` env var)
- This creates a second contract violation for testing alongside `GetOrderBroken`
- The OpenAPI spec declares `cancelled_at` as `type: string, format: date-time`; the inconsistent response is a deliberate contract violation

func TestGetOrder_ResponseSchema_AllRequiredFieldsPresent(t *testing.T) {
	srv := newTestServer(t)
	defer srv.Close()

	// Create first.
	createBody := `{"amount": 75, "currency": "USD", "description": "schema test"}`
	createResp, err := http.Post(srv.URL+"/orders", "application/json", bytes.NewBufferString(createBody))
	if err != nil {
		t.Fatalf("create order: %v", err)
	}
	defer createResp.Body.Close()
	var created map[string]any
	if err := json.NewDecoder(createResp.Body).Decode(&created); err != nil {
		t.Fatalf("decode create: %v", err)
	}
	id := created["id"].(string)

	// Get and check all declared required fields.
	resp, err := http.Get(srv.URL + "/orders/" + id)
	if err != nil {
		t.Fatalf("get order: %v", err)
	}
	defer resp.Body.Close()

	var order map[string]any
	if err := json.NewDecoder(resp.Body).Decode(&order); err != nil {
		t.Fatalf("decode response: %v", err)
	}

	requiredFields := []string{"id", "amount", "currency", "status", "created_at"}
	for _, field := range requiredFields {
		if order[field] == nil {
			t.Errorf("required field %q is absent or null in GetOrder response", field)
		}
	}

	// Type assertions for each field.
	if _, ok := order["id"].(string); !ok {
		t.Errorf("'id' must be a string, got %T", order["id"])
	}
	// JSON numbers decode as float64; amount must be a whole number.
	if amt, ok := order["amount"].(float64); !ok || amt != float64(int64(amt)) {
		t.Errorf("'amount' must be an integer, got %T: %v", order["amount"], order["amount"])
	}
	if _, ok := order["currency"].(string); !ok {
		t.Errorf("'currency' must be a string, got %T", order["currency"])
	}
	if status, ok := order["status"].(string); !ok {
		t.Errorf("'status' must be a string, got %T", order["status"])
	} else {
		validStatuses := map[string]bool{"pending": true, "confirmed": true, "shipped": true, "delivered": true, "cancelled": true}
		if !validStatuses[status] {
			t.Errorf("'status' value %q is not a declared enum value", status)
		}
	}
	if _, ok := order["created_at"].(string); !ok {
		t.Errorf("'created_at' must be a string, got %T", order["created_at"])
	}
}

// DoctorCheckResult mirrors the JSON output of bt doctor --output json.
type DoctorCheckResult struct {
	ID      string `json:"id"`
	Passed  bool   `json:"passed"`
	Level   string `json:"level"` // "pass", "warn", "fail"
	Message string `json:"message"`
}

type DoctorOutput struct {
	Checks []DoctorCheckResult `json:"checks"`

func btBinaryDoctor(t *testing.T) string {
	t.Helper()
	bin, err := exec.LookPath("bt")
	if err != nil {
		path := filepath.Join("..", "..", "..", "bt")
		if _, err := os.Stat(path); err == nil {
			return path
		}
		t.Skip("bt binary not found — run 'go build -o bt ./cmd/bt' first")
	}
	return bin
}

// TestDoctor_AllChecksHaveIDAndMessage verifies that every check result

	bt := btBinaryDoctor(t)

		"--config", "bt/backendtest.yaml",
		"--output", "json",

	// doctor exits non-zero if any check fails; we still want to parse the output.
	if err != nil {
		t.Logf("bt doctor exited with error (may be expected): %v", err)
	}

	var result DoctorOutput
	if jsonErr := json.Unmarshal(out, &result); jsonErr != nil {
		t.Fatalf("bt doctor --output json produced invalid JSON: %v\noutput: %s", jsonErr, out)

	if len(result.Checks) == 0 {

	for i, check := range result.Checks {
		if check.ID == "" {
			t.Errorf("check[%d]: ID must be non-empty", i)
		}
		if check.Message == "" {
			t.Errorf("check[%d] (%s): Message must be non-empty", i, check.ID)
		}

			t.Errorf("check[%d] (%s): Level must be non-empty", i, check.ID)
		}
	}
}

// ContractRunResult mirrors the JSON report output of bt run --strategy contract.
type ContractRunResult struct {
	Summary struct {
		Total       int `json:"total"`
		Passed      int `json:"passed"`
		Failed      int `json:"failed"`
		Quarantined int `json:"quarantined"`
	} `json:"summary"`
	Results []struct {
		OperationID string `json:"operation_id"`
		Passed      bool   `json:"passed"`
		Quarantined bool   `json:"quarantined"`
		Violations  []struct {
			Field    string `json:"field"`
			Expected string `json:"expected"`
			Actual   string `json:"actual"`
			Severity string `json:"severity"`
		} `json:"violations"`
		SchemaPath string `json:"schema_path"`
	} `json:"results"`
}

func btBinaryContract(t *testing.T) string {
	t.Helper()
	bin, err := exec.LookPath("bt")
	if err != nil {
		path := filepath.Join("..", "..", "..", "bt")

		}

// TestContractRun_JSONOutputIsValidSchema verifies the contract run produces
// a JSON report that matches the declared output schema — not just that it exits 0.
func TestContractRun_JSONOutputIsValidSchema(t *testing.T) {
	bt := btBinaryContract(t)

	out, err := exec.Command(bt, "run",
		"--config", "bt/backendtest.yaml",
		"--strategy", "contract",
		"--output", "json",
	).Output()
	if err != nil {
		t.Logf("bt run exited non-zero (may be expected for quarantined cases): %v", err)
	}

	var result ContractRunResult
	if jsonErr := json.Unmarshal(out, &result); jsonErr != nil {
		t.Fatalf("bt run --output json produced invalid JSON: %v\noutput: %s", jsonErr, out)
	}

	// Summary fields must all be non-negative and internally consistent.
	s := result.Summary
	if s.Total < 0 {
		t.Errorf("summary.total must be >= 0, got %d", s.Total)
	}
	if s.Passed+s.Failed+s.Quarantined != s.Total {
		t.Errorf("passed(%d) + failed(%d) + quarantined(%d) must equal total(%d)",
			s.Passed, s.Failed, s.Quarantined, s.Total)
	}
}

// TestContractRun_GetOrderBrokenIsQuarantined verifies the baseline quarantine model:

	out, _ := exec.Command(bt, "run",
		"--config", "bt/backendtest.yaml",
		"--strategy", "contract",

	if err := json.Unmarshal(out, &result); err != nil {
		t.Fatalf("invalid JSON: %v", err)
	}

				t.Error("GetOrderBroken must be marked quarantined in the report")

			// Each violation must have a populated field path.
			for j, v := range r.Violations {
				if v.Field == "" {
					t.Errorf("GetOrderBroken violation[%d]: Field must be non-empty", j)
				}
			}
			return
		}
	}
	t.Error("GetOrderBroken not found in contract run results")
}

// TestContractRun_WellFormedOperationsPass verifies the operations we expect
// to pass do pass — confirming the contract strategy does not produce false positives.
func TestContractRun_WellFormedOperationsPass(t *testing.T) {
	bt := btBinaryContract(t)

	out, _ := exec.Command(bt, "run",
		"--config", "bt/backendtest.yaml",
		"--strategy", "contract",
		"--output", "json",
	).Output()

	var result ContractRunResult
	if err := json.Unmarshal(out, &result); err != nil {
		t.Fatalf("invalid JSON: %v", err)
	}

	expectedPass := map[string]bool{
		"GetHealth":   false,
		"ListOrders":  false,
		"CreateOrder": false,
		"GetOrder":    false,
	}

	for _, r := range result.Results {
		if _, expected := expectedPass[r.OperationID]; expected {
			expectedPass[r.OperationID] = true
			if !r.Passed && !r.Quarantined {
				// Collect the violations to give a useful failure message.
				t.Errorf("operation %s expected to pass but failed. Violations:", r.OperationID)
				for _, v := range r.Violations {
					t.Errorf("  field=%q expected=%q actual=%q severity=%s", v.Field, v.Expected, v.Actual, v.Severity)
				}
			}
		}
	}

	for opID, seen := range expectedPass {
		if !seen {
			t.Errorf("expected operation %q in contract results, not found", opID)
		}
	}
}

// TestContractRun_ExitCodeZeroWhenOnlyQuarantinedFailures verifies the exit code model:
// a run where only quarantined operations fail must exit 0.
func TestContractRun_ExitCodeZeroWhenOnlyQuarantinedFailures(t *testing.T) {
	bt := btBinaryContract(t)

	cmd := exec.Command(bt, "run",
		"--config", "bt/backendtest.yaml",
		"--strategy", "contract",
	)
	err := cmd.Run()

	// err == nil means exit 0.
	// If we get exit 1, it means a non-quarantined operation failed.
	if err != nil {
		if exitErr, ok := err.(*exec.ExitError); ok {
			code := exitErr.ExitCode()
			if code == 1 {
				t.Errorf("exit code 1: a non-quarantined contract violation was detected — only quarantined failures should be present in a clean run")
			} else {
				t.Errorf("unexpected exit code %d: %v", code, err)
			}
		} else {
			t.Errorf("unexpected error type: %v", err)
		}
	}
}
